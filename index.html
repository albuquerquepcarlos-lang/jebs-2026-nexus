const json = (data, status = 200) => new Response(JSON.stringify(data), { status, headers: { "content-type": "application/json; charset=utf-8", "cache-control": "no-store" } }); function safeParse(value, fallback = null) { try { return JSON.parse(value); } catch { return fallback; } } function mergeHistory(base = {}, incoming = {}) { const out = { ...base }; for (const [id, value] of Object.entries(incoming || {})) { if (!id) continue; const old = out[id]; if (!old) { out[id] = value; continue; } const oldStops = Array.isArray(old.stops) ? old.stops : []; const newStops = Array.isArray(value?.stops) ? value.stops : []; const oldSteps = Array.isArray(old.steps) ? old.steps : []; const newSteps = Array.isArray(value?.steps) ? value.steps : []; out[id] = { ...old, ...value, // Nunca apaga check-ins já registrados. stops: [...new Set([...oldStops, ...newStops])], // Nunca perde etapas já concluídas. steps: oldSteps.length || newSteps.length ? [0, 1, 2].map( i => Boolean(oldSteps[i]) || Boolean(newSteps[i]) ) : undefined, // Se já estava concluído, continua concluído. done: Boolean(old.done || value?.done) }; } return out; } async function readState(env) { const [trs, hist, conv] = await Promise.all([ env.DB.prepare( "SELECT id, data FROM transports ORDER BY id" ).all(), env.DB.prepare( "SELECT id, data FROM transport_history" ).all(), env.DB.prepare( "SELECT person_key, status FROM conv_status" ).all() ]); const transports = []; for (const row of trs.results || []) { const parsed = safeParse(row.data); if (parsed) { transports.push(parsed); } else { console.warn( "Registro inválido ignorado em transports:", row.id ); } } const history = {}; for (const row of hist.results || []) { const parsed = safeParse(row.data); if (parsed) { history[row.id] = parsed; } else { console.warn( "Registro inválido ignorado em transport_history:", row.id ); } } const convStatuses = {}; for (const row of conv.results || []) { convStatuses[row.person_key] = row.status; } return { transports, history, convStatuses, initialized: transports.length > 0 }; } /** * Executa um conjunto de comandos. * * Primeiro tenta como batch. * Se algum comando causar erro, tenta individualmente para descobrir * exatamente qual registro está causando o problema. */ async function executeStatements(env, statements, labels = []) { if (!statements.length) { return { success: true, attempted: 0, written: 0, errors: [] }; } try { await env.DB.batch(statements); return { success: true, attempted: statements.length, written: statements.length, errors: [] }; } catch (batchError) { console.error("Erro no batch D1:", batchError); const errors = []; let written = 0; for (let i = 0; i < statements.length; i++) { try { await statements[i].run(); written++; } catch (error) { errors.push({ index: i, id: labels[i] || null, error: error?.message || String(error) }); console.error( "Erro ao gravar registro:", labels[i] || i, error ); } } return { success: errors.length === 0, attempted: statements.length, written, errors, batchError: batchError?.message || String(batchError) }; } } async function writeState(env, payload) { const transports = Array.isArray(payload?.transports) ? payload.transports : []; const incomingHistory = payload?.history && typeof payload.history === "object" ? payload.history : {}; const convStatuses = payload?.convStatuses && typeof payload.convStatuses === "object" ? payload.convStatuses : {}; /* * ============================================================ * 1. HISTÓRICO EXISTENTE * ============================================================ * * Antes de gravar uma nova planilha, recuperamos os históricos * que possuem os mesmos IDs. * * Isso impede que uma nova importação apague check-ins. */ const historyIds = Object.keys(incomingHistory).filter(Boolean); const existingHistory = {}; if (historyIds.length) { const placeholders = historyIds.map(() => "?").join(","); const rows = await env.DB.prepare( `SELECT id, data FROM transport_history WHERE id IN (${placeholders})` ) .bind(...historyIds) .all(); for (const row of rows.results || []) { const parsed = safeParse(row.data); if (parsed) { existingHistory[row.id] = parsed; } } } const history = mergeHistory( existingHistory, incomingHistory ); /* * ============================================================ * 2. TRANSPORTES * ============================================================ */ const transportStatements = []; const transportLabels = []; for (const transport of transports) { if (!transport?.id) continue; transportStatements.push( env.DB.prepare( `INSERT INTO transports (id, data, updated_at) VALUES (?, ?, datetime('now')) ON CONFLICT(id) DO UPDATE SET data = excluded.data, updated_at = excluded.updated_at` ).bind( String(transport.id), JSON.stringify(transport) ) ); transportLabels.push(String(transport.id)); } const transportResult = await executeStatements( env, transportStatements, transportLabels ); /* * ============================================================ * 3. HISTÓRICO / CHECK-INS * ============================================================ */ const historyStatements = []; const historyLabels = []; for (const [id, value] of Object.entries(history)) { if (!id || !value) continue; historyStatements.push( env.DB.prepare( `INSERT INTO transport_history (id, data, updated_at) VALUES (?, ?, datetime('now')) ON CONFLICT(id) DO UPDATE SET data = excluded.data, updated_at = excluded.updated_at` ).bind( String(id), JSON.stringify(value) ) ); historyLabels.push(String(id)); } const historyResult = await executeStatements( env, historyStatements, historyLabels ); /* * ============================================================ * 4. STATUS DE CONVERSA * ============================================================ */ const convStatements = []; const convLabels = []; for (const [personKey, status] of Object.entries( convStatuses )) { if (!personKey) continue; convStatements.push( env.DB.prepare( `INSERT INTO conv_status (person_key, status, updated_at) VALUES (?, ?, datetime('now')) ON CONFLICT(person_key) DO UPDATE SET status = excluded.status, updated_at = excluded.updated_at` ).bind( String(personKey), String(status) ) ); convLabels.push(String(personKey)); } const convResult = await executeStatements( env, convStatements, convLabels ); /* * ============================================================ * 5. CONFIRMAÇÃO REAL DO BANCO * ============================================================ */ const countResult = await env.DB.prepare( "SELECT COUNT(*) AS total FROM transports" ).first(); const totalInDatabase = Number( countResult?.total || 0 ); const allSuccessful = transportResult.errors.length === 0 && historyResult.errors.length === 0 && convResult.errors.length === 0; /* * Se houve erro, NÃO fingimos que a importação foi bem-sucedida. */ if (!allSuccessful) { return { ok: false, message: "A gravação no banco encontrou um ou mais erros.", transportsReceived: transports.length, transportsWritten: transportResult.written, totalInDatabase, transportErrors: transportResult.errors, historyErrors: historyResult.errors, convStatusErrors: convResult.errors, batchErrors: { transports: transportResult.batchError || null, history: historyResult.batchError || null, convStatuses: convResult.batchError || null } }; } /* * ============================================================ * 6. DEVOLVE O ESTADO ATUAL DO SERVIDOR * ============================================================ */ const finalState = await readState(env); return { ...finalState, ok: true, sync: { transportsReceived: transports.length, transportsWritten: transportResult.written, transportsInDatabase: totalInDatabase, historyWritten: historyResult.written, convStatusesWritten: convResult.written } }; } export default { async fetch(request, env) { const url = new URL(request.url); try { /* * ======================================================== * HEALTH CHECK * ======================================================== */ if (url.pathname === "/api/health") { await env.DB.prepare("SELECT 1").first(); return json({ ok: true, service: "jebs-2026-nexus", database: "connected" }); } /* * ======================================================== * GET ESTADO * ======================================================== */ if ( url.pathname === "/api/state" && request.method === "GET" ) { return json( await readState(env) ); } /* * ======================================================== * POST ESTADO * ======================================================== */ if ( url.pathname === "/api/state" && request.method === "POST" ) { let payload; try { payload = await request.json(); } catch (error) { return json( { ok: false, error: "O corpo enviado não é um JSON válido.", details: error?.message || String(error) }, 400 ); } const result = await writeState( env, payload ); /* * Se writeState encontrou problema, retornamos HTTP 500. * Assim o aplicativo não poderá dizer que deu tudo certo. */ if (result?.ok === false) { return json( result, 500 ); } return json(result); } /* * ======================================================== * ARQUIVOS DO APP * ======================================================== */ return env.ASSETS.fetch(request); } catch (error) { console.error( "ERRO GERAL DO WORKER:", error ); return json( { ok: false, error: error?.message || String(error) }, 500 ); } } };
